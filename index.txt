        /**
         * 將字串或數字解析為純浮點數
         */
        function parseCleanNumber(val) {
            if (val === undefined || val === null || val === '') return NaN;
            if (typeof val === 'number') return val;
            const cleaned = String(val).replace(/[,%\s億口萬元點]/g, '').trim();
            return parseFloat(cleaned);
        }

        /**
         * 格式化數字顯示（保留小數或加上千分位）
         */
        function formatNumberDisplay(rawVal, numVal, options = {}) {
            const { forceSign = false, isPercent = false } = options;
            if (typeof rawVal === 'string' && rawVal.trim() !== '') {
                // 若原始字串已有 % 且為百分比欄位，直接回傳
                if (isPercent && !rawVal.includes('%') && !isNaN(numVal)) {
                    return `${rawVal.trim()}%`;
                }
                return rawVal.trim();
            }

            if (isNaN(numVal)) return '載入中...';

            const formatted = numVal.toLocaleString('en-US', {
                maximumFractionDigits: 2
            });

            const signPrefix = (forceSign && numVal > 0) ? '+' : '';
            const percentSuffix = isPercent ? '%' : '';
            return `${signPrefix}${formatted}${percentSuffix}`;
        }

        /**
         * 從物件中依序尋找符合 keys 陣列的值，同時支援巢狀物件格式 { value, prev, diff }
         */
        function extractMetricData(payload, item) {
            let rawMain = undefined;
            for (const key of item.keys) {
                if (Object.prototype.hasOwnProperty.call(payload, key)) {
                    rawMain = payload[key];
                    break;
                }
            }

            // 支援物件形式，例如: "台指結算": { "value": 23150, "prev": 23000 } 或 { "current": 23150, "diff": 150 }
            if (rawMain !== null && typeof rawMain === 'object' && !Array.isArray(rawMain)) {
                const current = rawMain.value ?? rawMain.current ?? rawMain.today;
                const prev = rawMain.prev ?? rawMain.previous ?? rawMain.yesterday;
                const diff = rawMain.diff ?? rawMain.change;
                return { current, prev, diff };
            }

            // 支援扁平化鍵值，例如: "台指結算": 23150, "台指結算_前日": 23000 或 "台指結算_增減": 150
            let prev = undefined;
            if (item.prevKeys) {
                for (const pKey of item.prevKeys) {
                    if (Object.prototype.hasOwnProperty.call(payload, pKey)) {
                        prev = payload[pKey];
                        break;
                    }
                }
            }

            let diff = undefined;
            if (item.diffKeys) {
                for (const dKey of item.diffKeys) {
                    if (Object.prototype.hasOwnProperty.call(payload, dKey)) {
                        diff = payload[dKey];
                        break;
                    }
                }
            }

            return { current: rawMain, prev, diff };
        }

        /**
         * 依據規則計算燈號 ('red' | 'green' | 'gray')、主顯示文字、以及增減副標文字
         */
        function evaluateMetric(item, extracted) {
            const { current, prev, diff } = extracted;

            if (current === undefined || current === null || current === '') {
                return {
                    isLoaded: false,
                    light: 'gray',
                    displayText: '載入中...',
                    diffText: '',
                    diffColorClass: 'text-slate-500'
                };
            }

            const numVal = parseCleanNumber(current);
            const numPrev = parseCleanNumber(prev);
            let numDiff = parseCleanNumber(diff);

            // 若未直接提供 diff，但有提供 prev 與 current，則自動計算 (今日 - 前日)
            if (isNaN(numDiff) && !isNaN(numVal) && !isNaN(numPrev)) {
                numDiff = Number((numVal - numPrev).toFixed(2));
            }

            let light = 'gray';
            let diffText = '';
            let diffColorClass = 'text-slate-400';

            switch (item.ruleType) {
                case 'diff_up_red': {
                    // 1. 台指結算 & 8. 外資台指期未平倉：比前一天多亮紅燈，比前一天少亮綠燈
                    if (!isNaN(numDiff)) {
                        if (numDiff > 0) light = 'red';
                        else if (numDiff < 0) light = 'green';
                        else light = 'gray';

                        const arrow = numDiff > 0 ? '▲' : (numDiff < 0 ? '▼' : '—');
                        const sign = numDiff > 0 ? '+' : '';
                        const unit = item.diffUnit ? ` ${item.diffUnit}` : '';
                        diffText = `較前日 ${arrow} ${sign}${numDiff.toLocaleString('en-US', { maximumFractionDigits: 2 })}${unit}`;
                        diffColorClass = light === 'red' ? 'text-rose-400' : (light === 'green' ? 'text-emerald-400' : 'text-slate-400');
                    } else {
                        diffText = '較前日：無前日資料';
                    }
                    break;
                }

                case 'diff_down_red': {
                    // 3. 融資餘額(億)：比前一天少亮紅燈，比前一天多亮綠燈
                    if (!isNaN(numDiff)) {
                        if (numDiff < 0) light = 'red';
                        else if (numDiff > 0) light = 'green';
                        else light = 'gray';

                        const arrow = numDiff > 0 ? '▲' : (numDiff < 0 ? '▼' : '—');
                        const sign = numDiff > 0 ? '+' : '';
                        const unit = item.diffUnit ? ` ${item.diffUnit}` : '';
                        diffText = `較前日 ${arrow} ${sign}${numDiff.toLocaleString('en-US', { maximumFractionDigits: 2 })}${unit}`;
                        diffColorClass = light === 'red' ? 'text-rose-400' : (light === 'green' ? 'text-emerald-400' : 'text-slate-400');
                    } else {
                        diffText = '較前日：無前日資料';
                    }
                    break;
                }

                case 'threshold_8000': {
                    // 2. 現貨量(億)：低於 8000 億亮綠燈，超過 8000 億亮紅燈
                    if (!isNaN(numVal)) {
                        if (numVal > 8000) {
                            light = 'red';
                            diffText = '大於 8,000 億門檻';
                            diffColorClass = 'text-rose-400/90';
                        } else if (numVal < 8000) {
                            light = 'green';
                            diffText = '低於 8,000 億門檻';
                            diffColorClass = 'text-emerald-400/90';
                        } else {
                            light = 'gray';
                            diffText = '持平 8,000 億門檻';
                        }
                    }
                    break;
                }

                case 'pos_red': {
                    // 5,6,7 三大法人現貨 / 9,10 五大十大淨額 / 15 台積電ADR：正數亮紅燈，負數亮綠燈
                    if (!isNaN(numVal)) {
                        if (numVal > 0) light = 'red';
                        else if (numVal < 0) light = 'green';
                        else light = 'gray';
                    }
                    break;
                }

                case 'pos_green': {
                    // 13. 小台散戶多空比(%) & 14. 微台散戶多空比(%)：正數亮綠燈，負數亮紅燈
                    if (!isNaN(numVal)) {
                        if (numVal > 0) light = 'green';
                        else if (numVal < 0) light = 'red';
                        else light = 'gray';
                    }
                    break;
                }

                case 'neutral':
                default: {
                    // 4. 美元兌台幣 / 11. P/C未平倉比 / 12. 外資選擇權未平倉淨額(萬)：灰燈
                    light = 'gray';
                    break;
                }
            }

            const shouldForceSign = (item.ruleType === 'pos_red' || item.ruleType === 'pos_green');
            const displayText = formatNumberDisplay(current, numVal, {
                forceSign: shouldForceSign,
                isPercent: Boolean(item.isPercent)
            });

            return {
                isLoaded: true,
                light,
                displayText,
                diffText,
                diffColorClass
            };
        }

        /**
         * 讀取同目錄下的 data.json 並更新畫面
         */
        async function fetchChipData() {
            const refreshIcon = document.getElementById('refresh-icon');
            const errorBanner = document.getElementById('error-banner');

            try {
                // 加上時間戳記防止瀏覽器快取舊檔案
                const response = await fetch(`data.json?t=${new Date().getTime()}`);
                if (!response.ok) {
                    throw new Error(`HTTP 狀態碼: ${response.status}`);
                }

                const data = await response.json();
                const payload = data.metrics || data.items || data;

                // 更新頂部最後更新時間
                const updatedTime = data.last_updated || data.updated_at || data['最後更新時間'] || new Date().toLocaleString('zh-TW', { hour12: false });
                lastUpdatedEl.textContent = updatedTime;

                // 依序計算 15 項指標並更新卡片 UI
                METRICS_CONFIG.forEach((item) => {
                    const valueEl = document.getElementById(`value-${item.id}`);
                    const lightEl = document.getElementById(`light-${item.id}`);
                    const diffEl = document.getElementById(`diff-${item.id}`);

                    const extracted = extractMetricData(payload, item);
                    const result = evaluateMetric(item, extracted);
                    const style = LIGHT_STYLES[result.light] || LIGHT_STYLES.gray;

                    if (result.isLoaded) {
                        valueEl.classList.remove('animate-pulse', 'text-slate-400', 'text-rose-500', 'text-emerald-400', 'text-slate-100');
                        valueEl.classList.add(style.text);
                        valueEl.textContent = result.displayText;

                        lightEl.className = `inline-block w-3.5 h-3.5 rounded-full transition-all duration-300 ${style.dot}`;
                        lightEl.title = style.title;

                        diffEl.textContent = result.diffText;
                        diffEl.className = `text-xs font-medium min-h-[1.125rem] ${result.diffColorClass}`;
                    }
                });

            } catch (error) {
                console.error('讀取 data.json 失敗:', error);
                errorBanner.classList.remove('hidden');
            } finally {
                setTimeout(() => {
                    refreshIcon.classList.remove('animate-spin');
                }, 400);
            }
        }

        // 頁面載入後初始化骨架並抓取 data.json
        window.addEventListener('DOMContentLoaded', () => {
            initCardsSkeleton();
            fetchChipData();
        });
    </script>
